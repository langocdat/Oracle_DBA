# Required
- The target oracle home version same to source oracle home
- The target os same to source os

# Prepare
## Step 1.1: Copy ra_wallet from source database to target database
*Mục tiêu: Sao chép thư mục ra_wallet sang máy chủ cần khôi phục*
```
[oracle]$ cd $ORACLE_HOME/bin
[oracle]$ tar -cvzf ra_wallet.tar.gz ra_wallet
[oracle]$ scp ra_wallet.tar.gz oracle@68.68.xxx.xxx:$ORACLE_HOME/dbs
```

## Step 1.2: Create the configuration ZDLRA file on target database
*Chú ý: Thay đổi giá trị ZDLRA_$DB_NAME phù hợp*
```
[oracle]$ cd $ORACLE_HOME/dbs/
[oracle]$ vi ra$ORACLE_SID.ora
RA_WALLET='location=file:/u01/app/oracle/product/19.3.0/db_1/dbs/ra_wallet credential_alias= ZDLRA_$DB_NAME'
```

## Step 1.3: Add alias in tnsnames.ora file on target database
*Chú ý: Thay đổi giá trị ZDLRA_$DB_NAME phù hợp*
```
[oracle]$ cd $ORACLE_HOME/network/admin
[oracle]$ vi tnsnames.ora
ZDLRA_$DB_NAME  =
  (DESCRIPTION =
    (ADDRESS_LIST =
      (ADDRESS = (PROTOCOL = TCP)(HOST =68.68.xxx.xxx)(PORT = 15xx))
    )
    (CONNECT_DATA =
      (SERVICE_NAME = db1db1)
    )
  )
```

## Step 1.4: Configure in sqlnet.ora file on target database
```
[oracle]$ cd $ORACLE_HOME/network/admin
[oracle]$ vi sqlnet.ora

SQLNET.WALLET_OVERRIDE = true
WALLET_LOCATION = 
(SOURCE = 
    (METHOD = FILE)
    (METHOD_DATA =
      (DIRECTORY = /u01/app/oracle/product/19.0.0.0/dbhome_1/dbs/ra_wallet)
   )
)
```

# Restore database from ZDLRA machine
## 2.1: Startup nomount
```
[oracle]$ vi init_<SID>.ora
db_name = $DB_NAME

[oracle]$ sqlplus / as sysdba
SQL> startup nomount
```

# Step 2.2 : Restore spfile from ZDLRA
```
[oracle]$ rman target / catalog=/@ZDLRA_$DB_NAME 
RMAN> set dbid=2500287482
RMAN> 
run
{ 
allocate channel ch1 device type 'SBT_TAPE' PARMS 'SBT_LIBRARY=$ORACLE_HOME/lib/libra.so,
ENV=(RA_FORMAT=true,RA_WALLET=location=file:$ORACLE_HOME/ra_wallet credential_alias=ZDLRA_$DB_NAME)';
restore spfile to '$ORACLE_HOME/dbs/spfileCDBC06_case.ora' from autobackup;
}
```

# Step 2.3: Restart with spfile
```
SQL> shutdown immediate;
SQL> startup nomount;
```

# Step 2.4: Restore controlfile from ZDLRA
```
[oracle]$ rman target / catalog=/@ZDLRA_$DB_NAME 
RMAN> list backup of controlfile summary;
RMAN>
run 
{
allocate channel ch1 device type 'SBT_TAPE' PARMS 'SBT_LIBRARY=/u01/app/oracle/product/19.3.0/db_1/lib/libra.so, ENV=(RA_WALLET=location=file:/u01/app/oracle/product/19.3.0/db_1/dbs/ra_wallet credential_alias=ZDLRA_$DB_NAME)'; restore controlfile from autobackup; 
}
```

#  Step 2.5: Mount database
```
RMAN> alter database mount;
```

# Step 2.6: Restore database and recover database
```
RMAN> 
run {
allocate channel ch1 device type 'SBT_TAPE' PARMS 'SBT_LIBRARY=$ORACLE_HOME/lib/libra.so,
ENV=(RA_FORMAT=true,RA_WALLET=location=file:$ORACLE_HOME/ra_wallet credential_alias=ZDLRA_$DB_NAME)';
allocate channel ch2 device type 'SBT_TAPE' PARMS 'SBT_LIBRARY=$ORACLE_HOME/lib/libra.so,
ENV=(RA_FORMAT=true,RA_WALLET=location=file:$ORACLE_HOME/ra_wallet credential_alias=ZDLRA_$DB_NAME)';
allocate channel ch3 device type 'SBT_TAPE' PARMS 'SBT_LIBRARY=$ORACLE_HOME/lib/libra.so,
ENV=(RA_FORMAT=true,RA_WALLET=location=file:$ORACLE_HOME/ra_wallet credential_alias=ZDLRA_$DB_NAME)';
allocate channel ch4 device type 'SBT_TAPE' PARMS 'SBT_LIBRARY=$ORACLE_HOME/lib/libra.so,
ENV=(RA_FORMAT=true,RA_WALLET=location=file:$ORACLE_HOME/ra_wallet credential_alias=ZDLRA_$DB_NAME)';
allocate channel ch5 device type 'SBT_TAPE' PARMS 'SBT_LIBRARY=$ORACLE_HOME/lib/libra.so,
ENV=(RA_FORMAT=true,RA_WALLET=location=file:$ORACLE_HOME/ra_wallet credential_alias=ZDLRA_$DB_NAME)';
allocate channel ch6 device type 'SBT_TAPE' PARMS 'SBT_LIBRARY=$ORACLE_HOME/lib/libra.so,
ENV=(RA_FORMAT=true,RA_WALLET=location=file:$ORACLE_HOME/ra_wallet credential_alias=ZDLRA_$DB_NAME)';
allocate channel ch7 device type 'SBT_TAPE' PARMS 'SBT_LIBRARY=$ORACLE_HOME/lib/libra.so,
ENV=(RA_FORMAT=true,RA_WALLET=location=file:$ORACLE_HOME/ra_wallet credential_alias=ZDLRA_$DB_NAME)';
allocate channel ch8 device type 'SBT_TAPE' PARMS 'SBT_LIBRARY=$ORACLE_HOME/lib/libra.so,
ENV=(RA_FORMAT=true,RA_WALLET=location=file:$ORACLE_HOME/ra_wallet credential_alias=ZDLRA_$DB_NAME)';
set newname for database to '+DATA';
restore database;
switch datafile all;
recover database;
}
```

*Chú ý: Nếu xuất hiện lỗi khi chạy đoạn lệnh trên, cần bỏ tham số RA_FORMAT=true trên từng dòng allocate channel. Ví dụ:*
```
allocate channel ch8 device type 'SBT_TAPE' PARMS 'SBT_LIBRARY=/u01/app/oracle/product/19.3.0/db_1/lib/libra.so,
ENV=(RA_WALLET=location=file:/u01/app/oracle/product/19.3.0/db_1/dbs/ra_wallet credential_alias=ZDLRA_$DB_NAME)';
```
*Nguyên nhân: Bản backup gốc không sử dụng RA_FORMAT=true*

# Step 2.7: Set parameter
```
[oracle]$ sqlplus / as sysdba
SQL> alter system set db_recovery_dest_size = '500G' scope=both sid='*';
SQL> alter system set db_recovery_dest = '+RECO' scope=both sid='*';
SQL> ALTER SYSTEM SET log_archive_dest_1 = 'LOCATION=USE_DB_RECOVERY_FILE_DEST' SCOPE=BOTH SID='*';
```

# Step 2.8: Open the database
```
SQL> alter database open;
Chú ý: Nếu không open được, cần sử dụng thêm tham số resetlogs.
SQL> alter database open resetlogs;
```
